## 一、环境准备

### 1. 启动服务

```bash
# 进入项目目录
cd Kinlin_AI

# 启动所有服务（使用Docker）
docker-compose up -d

# 或手动启动Python服务
cd agent
python -m uvicorn app.main:app --host 0.0.0.0 --port 8000
```

### 2. 验证服务

访问 `http://localhost:8000/docs` 查看API文档

## 二、数字人功能

### 创建数字人

```python
import requests

# 创建律师角色的数字人
response = requests.post(
    "http://localhost:8000/ai/digital-human/create",
    json={
        "role_id": "lawyer_001",
        "personality": "严谨、专业",
        "profession": "律师",
        "style": "realistic"  # realistic/cartoon/anime
    }
)

avatar_data = response.json()["data"]
print(f"数字人创建成功: {avatar_data['role_id']}")
```

### 更新数字人动画

```python
# 当数字人说话时，更新动画
response = requests.post(
    "http://localhost:8000/ai/digital-human/animation",
    json={
        "role_id": "lawyer_001",
        "audio": "<base64编码的音频>",  # 实际使用时需要编码音频
        "text": "这是回复内容"
    }
)

animation = response.json()["data"]
print(f"动画数据: {animation}")
```

## 三、情感感知功能

### 情感分析

```python
# 分析用户情感
response = requests.post(
    "http://localhost:8000/ai/emotion/analyze",
    json={
        "text": "我最近很焦虑，工作压力很大",
        "audio_features": {
            "pitch": 0.6,
            "energy": 0.7,
            "tempo": 0.5
        }
    }
)

emotion = response.json()["data"]
print(f"用户情感: {emotion['emotion']}, 强度: {emotion['intensity']}")
```

### 情感感知回复

```python
# 生成情感感知的回复
response = requests.post(
    "http://localhost:8000/ai/emotion/response",
    json={
        "question": "我最近很焦虑",
        "base_role": {
            "role_id": "counselor",
            "personality": "温和、耐心",
            "knowledge_domain": ["心理", "情感"]
        },
        "text": "我最近很焦虑，工作压力很大"
    }
)

result = response.json()["data"]
print(f"回复: {result['text']}")
print(f"数字人情感: {result['emotion']}")
```

## 四、角色融合功能

### 融合多个角色的回答

```python
# 融合律师和商业顾问的回答
response = requests.post(
    "http://localhost:8000/ai/role-fusion/fuse",
    json={
        "question": "我想创业，需要法律和商业建议",
        "available_roles": [
            {
                "role_id": "lawyer",
                "knowledge_domain": ["法律", "合同"],
                "personality": "严谨、专业"
            },
            {
                "role_id": "business",
                "knowledge_domain": ["商业", "策略"],
                "personality": "务实、创新"
            }
        ],
        "role_responses": {
            "lawyer": "从法律角度，需要注意合同条款、知识产权保护...",
            "business": "从商业角度，建议考虑市场定位、商业模式..."
        }
    }
)

fused = response.json()["data"]
print(f"融合后的回答: {fused['response']}")
print(f"角色权重: {fused['weights']}")
```

## 五、知识图谱功能

### 构建知识图谱

```python
# 从文档构建知识图谱
response = requests.post(
    "http://localhost:8000/ai/knowledge-graph/build",
    json={
        "documents": [
            {
                "doc_id": "doc1",
                "text": "张三是一名律师，属于ABC律师事务所。李四是该事务所的合伙人。",
                "metadata": {}
            }
        ]
    }
)

stats = response.json()["data"]
print(f"实体数量: {stats['entities_count']}")
print(f"三元组数量: {stats['triples_count']}")
```

### 混合检索

```python
# 使用知识图谱增强检索
response = requests.post(
    "http://localhost:8000/ai/knowledge-graph/search",
    json={
        "question": "ABC律师事务所的律师有哪些？",
        "vector_db_results": [
            {"content": "...", "score": 0.8}
        ],
        "top_k": 5
    }
)

result = response.json()["data"]
print(f"知识图谱结果: {result['kg_results']}")
print(f"融合结果: {result['fused_results']}")
```

### 知识推理

```python
# 基于知识图谱进行推理
response = requests.post(
    "http://localhost:8000/ai/knowledge-graph/reason",
    json={
        "question": "张三和李四的关系是什么？"
    }
)

reasoning = response.json()["data"]
print(f"推理路径: {reasoning['reasoning_paths']}")
print(f"结论: {reasoning['conclusions']}")
```

## 七、完整示例

### 示例：情感感知的数字人对话

```python
import requests

BASE_URL = "http://localhost:8000/ai"

# 1. 创建数字人
print("创建数字人...")
response = requests.post(
    f"{BASE_URL}/digital-human/create",
    json={
        "role_id": "counselor_001",
        "personality": "温和、耐心",
        "profession": "心理咨询师",
        "style": "realistic"
    }
)
avatar = response.json()["data"]
print(f"✓ 数字人创建成功: {avatar['role_id']}")

# 2. 情感分析
print("\n分析用户情感...")
response = requests.post(
    f"{BASE_URL}/emotion/analyze",
    json={
        "text": "我最近很焦虑，工作压力很大，不知道该怎么办"
    }
)
emotion = response.json()["data"]
print(f"✓ 用户情感: {emotion['emotion']}, 强度: {emotion['intensity']:.2f}")

# 3. 生成情感感知回复
print("\n生成情感感知回复...")
response = requests.post(
    f"{BASE_URL}/emotion/response",
    json={
        "question": "我最近很焦虑，工作压力很大",
        "base_role": {
            "role_id": "counselor_001",
            "personality": "温和、耐心",
            "knowledge_domain": ["心理", "情感"]
        },
        "text": "我最近很焦虑，工作压力很大"
    }
)
result = response.json()["data"]
print(f"✓ 回复: {result['text']}")
print(f"✓ 数字人情感: {result['emotion']}")

# 4. 更新数字人动画
print("\n更新数字人动画...")
response = requests.post(
    f"{BASE_URL}/digital-human/animation",
    json={
        "role_id": "counselor_001",
        "audio": b"",  # 实际使用时需要真实音频
        "text": result['text']
    }
)
animation = response.json()["data"]
print(f"✓ 动画数据生成成功，时长: {animation['duration']:.2f}秒")
```

## 八、常见问题

### Q1: 如何启用情感感知？

A: 在对话API中添加 `enable_emotion_aware: true` 参数即可。

### Q2: 知识图谱需要预先构建吗？

A: 是的，需要先调用 `/knowledge-graph/build` 构建图谱，然后才能使用检索和推理功能。

### Q3: 数字人支持哪些风格？

A: 支持三种风格：`realistic`（写实）、`cartoon`（卡通）、`anime`（二次元）。

### Q4: 如何查看所有可用的API？

A: 启动服务后访问 `http://localhost:8000/docs` 查看完整的API文档。
