## 3. 状态管理设计

### 3.1 对话状态管理

```typescript
// stores/chat.ts
import { defineStore } from 'pinia'

export const useChatStore = defineStore('chat', {
  state: () => ({
    messages: [] as Message[],
    currentRole: null as Role | null,
    isLoading: false,
    contextId: null as string | null,
    inputText: '',
    isRecording: false,
    isPlaying: false
  }),
  
  getters: {
    lastMessage: (state) => state.messages[state.messages.length - 1],
    unreadCount: (state) => state.messages.filter(m => !m.read).length
  },
  
  actions: {
    async sendMessage(text: string) {
      this.isLoading = true
      try {
        const response = await chatApi.sendMessage({
          text,
          role: this.currentRole?.id,
          contextId: this.contextId
        })
        
        this.messages.push({
          type: 'user',
          content: text,
          timestamp: Date.now()
        })
        
        this.messages.push({
          type: 'assistant',
          content: response.text,
          animation: response.animation,
          timestamp: Date.now()
        })
        
        this.contextId = response.contextId
      } finally {
        this.isLoading = false
      }
    },
    
    clearHistory() {
      this.messages = []
      this.contextId = null
    }
  }
})
```

### 3.2 角色状态管理

```typescript
// stores/role.ts
export const useRoleStore = defineStore('role', {
  state: () => ({
    builtinRoles: [] as Role[],
    customRoles: [] as Role[],
    currentRole: null as Role | null,
    selectedRoleId: null as string | null
  }),
  
  actions: {
    async loadRoles() {
      const [builtin, custom] = await Promise.all([
        roleApi.getBuiltinRoles(),
        roleApi.getCustomRoles()
      ])
      this.builtinRoles = builtin
      this.customRoles = custom
    },
    
    async selectRole(roleId: string) {
      const role = [...this.builtinRoles, ...this.customRoles]
        .find(r => r.id === roleId)
      if (role) {
        this.currentRole = role
        this.selectedRoleId = roleId
      }
    },
    
    async createCustomRole(roleData: CreateRoleData) {
      const role = await roleApi.createRole(roleData)
      this.customRoles.push(role)
      return role
    }
  }
})
```

## 4. API接口设计

### 4.1 对话接口

```typescript
// services/api/chat.ts
export const chatApi = {
  // 发送文本消息
  async sendMessage(data: {
    text: string
    role?: string
    contextId?: string
  }): Promise<ChatResponse> {
    return request.post('/api/chat/text', data)
  },
  
  // 发送语音消息
  async sendVoice(data: {
    audio: Blob
    role?: string
    contextId?: string
  }): Promise<ChatResponse> {
    const formData = new FormData()
    formData.append('audio', data.audio)
    if (data.role) formData.append('role', data.role)
    if (data.contextId) formData.append('contextId', data.contextId)
    
    return request.post('/api/chat/voice', formData, {
      headers: { 'Content-Type': 'multipart/form-data' }
    })
  },
  
  // 获取对话历史
  async getHistory(contextId: string): Promise<Message[]> {
    return request.get(`/api/chat/history/${contextId}`)
  },
  
  // 清除对话历史
  async clearHistory(contextId: string): Promise<void> {
    return request.delete(`/api/chat/history/${contextId}`)
  }
}
```

## 5. 数字人渲染实现

### 5.1 渲染器初始化

```typescript
// utils/digital-human/renderer.ts
import * as THREE from 'three'

export class DigitalHumanRenderer {
  private scene: THREE.Scene
  private camera: THREE.PerspectiveCamera
  private renderer: THREE.WebGLRenderer
  private digitalHuman: DigitalHumanModel | null = null
  
  constructor(canvas: HTMLCanvasElement) {
    // 初始化场景
    this.scene = new THREE.Scene()
    this.scene.background = new THREE.Color(0xf0f0f0)
    
    // 初始化相机
    this.camera = new THREE.PerspectiveCamera(
      45,
      canvas.width / canvas.height,
      0.1,
      1000
    )
    this.camera.position.set(0, 1.6, 3)
    
    // 初始化渲染器
    this.renderer = new THREE.WebGLRenderer({
      canvas,
      antialias: true,
      alpha: true
    })
    this.renderer.setSize(canvas.width, canvas.height)
    this.renderer.setPixelRatio(window.devicePixelRatio)
    
    // 添加灯光
    this.setupLights()
    
    // 开始渲染循环
    this.animate()
  }
  
  private setupLights() {
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.6)
    this.scene.add(ambientLight)
    
    const directionalLight = new THREE.DirectionalLight(0xffffff, 0.8)
    directionalLight.position.set(5, 10, 5)
    this.scene.add(directionalLight)
  }
  
  async loadRole(role: Role) {
    // 加载数字人模型
    const loader = new GLTFLoader()
    const model = await loader.loadAsync(role.avatarModelPath)
    
    this.digitalHuman = new DigitalHumanModel(model.scene)
    this.scene.add(this.digitalHuman.mesh)
  }
  
  playAnimation(animationName: string) {
    if (this.digitalHuman) {
      this.digitalHuman.playAnimation(animationName)
    }
  }
  
  private animate() {
    requestAnimationFrame(() => this.animate())
    
    if (this.digitalHuman) {
      this.digitalHuman.update()
    }
    
    this.renderer.render(this.scene, this.camera)
  }
  
  dispose() {
    this.renderer.dispose()
    this.digitalHuman?.dispose()
  }
}
```
