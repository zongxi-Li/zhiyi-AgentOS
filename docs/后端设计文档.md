### 1.2 技术选型

#### 核心框架
- **Python 3.9+** - 主编程语言
- **FastAPI** - Web框架（高性能、异步支持）
- **Uvicorn** - ASGI服务器
- **Pydantic** - 数据验证

#### AI框架
- **麒麟AI SDK** - 核心AI能力
- **PyTorch** / **TensorFlow** - 深度学习框架（如需要）
- **Transformers** - 预训练模型库

#### 数据库
- **PostgreSQL** - 主数据库（关系型数据）
- **Redis** - 缓存和会话存储
- **MongoDB** - 文档存储（可选，用于非结构化数据）

#### 消息队列
- **RabbitMQ** / **Redis Queue** - 异步任务队列

#### 其他工具
- **SQLAlchemy** - ORM框架
- **Alembic** - 数据库迁移
- **Celery** - 异步任务处理
- **WebSocket** - 实时通信

## 2. 麒麟AI SDK集成

### 2.1 SDK初始化

```python
# app/ai_engine/kylin_sdk/client.py
from kylin_ai_sdk import KylinAIClient
from app.config import settings

class KylinSDKClient:
    """麒麟AI SDK客户端封装"""
    
    def __init__(self):
        self.client = KylinAIClient(
            api_key=settings.KYLIN_AI_API_KEY,
            api_endpoint=settings.KYLIN_AI_ENDPOINT,
            timeout=settings.KYLIN_AI_TIMEOUT
        )
        self._initialize_models()
    
    def _initialize_models(self):
        """初始化模型"""
        # 文本生成模型
        self.text_model = self.client.get_model(
            model_type="text_generation",
            model_name=settings.TEXT_MODEL_NAME
        )
        
        # 语音识别模型
        self.asr_model = self.client.get_model(
            model_type="speech_recognition",
            model_name=settings.ASR_MODEL_NAME
        )
        
        # 语音合成模型
        self.tts_model = self.client.get_model(
            model_type="speech_synthesis",
            model_name=settings.TTS_MODEL_NAME
        )
    
    async def generate_text(
        self,
        prompt: str,
        context: list = None,
        role_config: dict = None,
        **kwargs
    ) -> dict:
        """
        文本生成
        
        Args:
            prompt: 输入提示
            context: 对话上下文
            role_config: 角色配置
            **kwargs: 其他参数（temperature, top_p等）
        
        Returns:
            {
                "text": "生成的文本",
                "confidence": 0.95,
                "tokens_used": 150
            }
        """
        # 构建请求参数
        request_params = {
            "prompt": prompt,
            "max_tokens": kwargs.get("max_tokens", 512),
            "temperature": kwargs.get("temperature", 0.7),
            "top_p": kwargs.get("top_p", 0.9),
        }
        
        # 添加上下文
        if context:
            request_params["context"] = context
        
        # 添加角色配置
        if role_config:
            request_params["system_prompt"] = role_config.get("system_prompt")
            request_params["style"] = role_config.get("style")
        
        # 调用SDK
        response = await self.text_model.generate_async(**request_params)
        
        return {
            "text": response.text,
            "confidence": response.confidence,
            "tokens_used": response.tokens_used,
            "finish_reason": response.finish_reason
        }
    
    async def recognize_speech(
        self,
        audio_data: bytes,
        language: str = "zh-CN",
        **kwargs
    ) -> dict:
        """
        语音识别
        
        Args:
            audio_data: 音频数据（字节流）
            language: 语言代码
            **kwargs: 其他参数
        
        Returns:
            {
                "text": "识别的文本",
                "confidence": 0.92,
                "duration": 3.5
            }
        """
        response = await self.asr_model.recognize_async(
            audio_data=audio_data,
            language=language,
            **kwargs
        )
        
        return {
            "text": response.text,
            "confidence": response.confidence,
            "duration": response.duration
        }
    
    async def synthesize_speech(
        self,
        text: str,
        voice: str = "default",
        speed: float = 1.0,
        pitch: float = 1.0,
        **kwargs
    ) -> bytes:
        """
        语音合成
        
        Args:
            text: 要合成的文本
            voice: 语音类型
            speed: 语速
            pitch: 音调
            **kwargs: 其他参数
        
        Returns:
            音频数据（字节流）
        """
        response = await self.tts_model.synthesize_async(
            text=text,
            voice=voice,
            speed=speed,
            pitch=pitch,
            **kwargs
        )
        
        return response.audio_data
```

```python
from app.ai_engine.kylin_sdk.client import KylinSDKClient
from app.models.role import Role
from app.utils.cache import cache_manager

class AIService:
    """AI服务封装"""
    
    def __init__(self):
        self.sdk_client = KylinSDKClient()
        self.cache = cache_manager
    
    async def chat(
        self,
        message: str,
        role: Role = None,
        context_id: str = None,
        **kwargs
    ) -> dict:
        """
        对话服务
        
        Args:
            message: 用户消息
            role: 角色对象
            context_id: 上下文ID
            **kwargs: 其他参数
        
        Returns:
            对话响应
        """
        # 获取对话上下文
        context = []
        if context_id:
            context = await self._get_context(context_id)
        
        # 构建角色配置
        role_config = None
        if role:
            role_config = {
                "system_prompt": role.system_prompt,
                "style": role.dialogue_style,
                "personality": role.personality
            }
        
        # 检查缓存
        cache_key = self._generate_cache_key(message, role, context)
        cached_response = await self.cache.get(cache_key)
        if cached_response:
            return cached_response
        
        # 调用SDK生成回复
        response = await self.sdk_client.generate_text(
            prompt=message,
            context=context,
            role_config=role_config,
            **kwargs
        )
        
        # 缓存响应
        await self.cache.set(cache_key, response, ttl=3600)
        
        # 保存对话历史
        if context_id:
            await self._save_message(context_id, message, response["text"])
        
        return response
    
    async def speech_to_text(
        self,

    ) -> bytes:
        """文本转语音"""
        return await self.sdk_client.synthesize_speech(
            text=text,
            voice=voice,
            **kwargs
        )
    
    async def _get_context(self, context_id: str) -> list:
        """获取对话上下文"""
        # 从数据库或缓存获取
        pass
    
    async def _save_message(
        self,
        context_id: str,
        user_message: str,
        assistant_message: str
    ):
        """保存对话消息"""
        # 保存到数据库
        pass
    
    def _generate_cache_key(
        self,
        message: str,
        role: Role = None,
        context: list = None
    ) -> str:
        """生成缓存键"""
        import hashlib
        key_parts = [message]
        if role:
            key_parts.append(str(role.id))
        if context:
            key_parts.append(str(context))
        key_string = "|".join(key_parts)
        return hashlib.md5(key_string.encode()).hexdigest()
```

### 3.2 WebSocket API

```python
# app/api/websocket.py
from fastapi import WebSocket, WebSocketDisconnect
from app.services.chat_service import ChatService

class ConnectionManager:
    """WebSocket连接管理"""
    
    def __init__(self):
        self.active_connections: list[WebSocket] = []
    
    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)
    
    def disconnect(self, websocket: WebSocket):
        self.active_connections.remove(websocket)
    
    async def send_personal_message(self, message: dict, websocket: WebSocket):
        await websocket.send_json(message)
    
    async def broadcast(self, message: dict):
        for connection in self.active_connections:
            await connection.send_json(message)

manager = ConnectionManager()

@app.websocket("/ws/chat")
async def websocket_chat(websocket: WebSocket):
    await manager.connect(websocket)
    chat_service = ChatService()
    
    try:
        while True:
            data = await websocket.receive_json()
            
            # 处理消息
            if data["type"] == "message":
                response = await chat_service.send_message(
                    text=data["text"],
                    role_id=data.get("role_id"),
                    context_id=data.get("context_id")
                )
                
                # 发送回复
                await manager.send_personal_message({
                    "type": "response",
                    "text": response["text"],
                    "animation": response.get("animation")
                }, websocket)
            
            elif data["type"] == "voice":
                # 处理语音消息
                audio_data = data["audio"]
                response = await chat_service.send_voice_message(
                    audio_data=audio_data,
                    role_id=data.get("role_id")
                )
                
                await manager.send_personal_message({
                    "type": "response",
                    "text": response["text"],
                    "audio": response.get("audio")
                }, websocket)
    
    except WebSocketDisconnect:
        manager.disconnect(websocket)
```

```python
# app/services/chat_service.py
from app.services.ai_service import AIService
from app.models.chat import Conversation, Message
from app.database.session import get_db

class ChatService:
    """对话服务"""
    
    def __init__(self):
        self.ai_service = AIService()
        self.db = next(get_db())
    
    async def send_message(
        self,
        text: str,
        role_id: str = None,
        context_id: str = None,
        user_id: str = None
    ) -> dict:
        """发送消息并获取回复"""
        
        # 获取或创建对话
        conversation = await self._get_or_create_conversation(
            context_id, user_id, role_id
        )
        
        # 保存用户消息
        user_message = Message(
            conversation_id=conversation.id,
            role="user",
            content=text
        )
        self.db.add(user_message)
        
        # 获取AI回复
        ai_response = await self.ai_service.chat(
            message=text,
            role_id=role_id,
            context_id=conversation.context_id
        )
        
        # 保存AI回复
        assistant_message = Message(
            conversation_id=conversation.id,
            role="assistant",
            content=ai_response["text"],
            metadata={"confidence": ai_response["confidence"]}
        )
        self.db.add(assistant_message)
        self.db.commit()
        
        return {
            "text": ai_response["text"],
            "context_id": conversation.context_id,
            "confidence": ai_response["confidence"]
        }
    
    async def _get_or_create_conversation(
        self,
        context_id: str,
        user_id: str,
        role_id: str
    ) -> Conversation:
        """获取或创建对话"""
        if context_id:
            conversation = self.db.query(Conversation).filter(
                Conversation.context_id == context_id
            ).first()
            if conversation:
                return conversation
        
        # 创建新对话
        conversation = Conversation(
            user_id=user_id,
            role_id=role_id,
            context_id=context_id or str(uuid.uuid4())
        )
        self.db.add(conversation)
        self.db.commit()
        return conversation
```

```python
from app.ai_engine.federated_learning import FederatedLearningEngine

class FederatedLearningService:
    """联邦学习服务"""
    
    def __init__(self):
        self.fl_engine = FederatedLearningEngine()
    
    async def train_local_model(
        self,
        user_id: str,
        local_data: list,
        global_model: dict
    ) -> dict:
        """本地模型训练"""
        local_updates = await self.fl_engine.train_local(
            data=local_data,
            base_model=global_model
        )
        
        # 加密参数
        encrypted_updates = self._encrypt_updates(local_updates)
        
        return {
            "user_id": user_id,
            "encrypted_updates": encrypted_updates,
            "data_size": len(local_data)
        }
    
    async def aggregate_models(
        self,
        local_updates: list[dict]
    ) -> dict:
        """聚合模型参数"""
        return await self.fl_engine.aggregate(local_updates)
```
